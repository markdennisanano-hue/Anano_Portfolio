function openModal(title, files) { document.getElementById('modalTitle').innerText = title; const fileList = document.getElementById('fileList'); fileList.innerHTML = ''; files.forEach(file => { const fileItem = document.createElement('div'); fileItem.className = 'file-item'; const ext = file.split('.').pop().toLowerCase(); if (['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext)) { const img = document.createElement('img'); img.src = file; img.className = 'file-preview'; fileItem.appendChild(img); } const nameLabel = document.createElement('span'); nameLabel.innerText = file; nameLabel.style.fontWeight = '600'; fileItem.appendChild(nameLabel); const btnGroup = document.createElement('div'); btnGroup.className = 'btn-group'; const viewBtn = document.createElement('a'); viewBtn.href = file; viewBtn.target = '_blank'; viewBtn.className = 'btn btn-view'; viewBtn.innerText = 'View'; const downloadBtn = document.createElement('a'); downloadBtn.href = file; downloadBtn.download = file; downloadBtn.className = 'btn btn-download'; downloadBtn.innerText = 'Download'; btnGroup.appendChild(viewBtn); btnGroup.appendChild(downloadBtn); fileItem.appendChild(btnGroup); fileList.appendChild(fileItem); }); document.getElementById('fileModal').style.display = 'flex'; } function closeModal() { document.getElementById('fileModal').style.display = 'none'; } window.onclick = function(event) { const modal = document.getElementById('fileModal'); if (event.target == modal) { modal.style.display = 'none'; } }
